4. RUN A TARGET (the normal way)
------------------------------------------------------------------------

./venv/bin/python -m targets.run --list                        # what can I test?

#   wikipedia         Wikipedia, read-only
#   contacts-app      Android Contacts (needs the emulator)

./venv/bin/python -u -m targets.run dataghurhi-auth --ingest > logs/dataghurhi_auth_run.log 2>&1

# Only one batch can run at a time (web_player/runlock.py refuses a second one).

5. SIGNED-IN SESSION (for dataghurhi-auth)
------------------------------------------------------------------------

# Opens a browser. Log in by hand (solve the reCAPTCHA), then CLOSE THE WINDOW —
# the session is written to auth.json only when the window closes.
./venv/bin/python -m playwright codegen --save-storage=auth.json https://dataghurhi.cse.buet.ac.bd

# Check the session still works (prints signed in: True)
./venv/bin/python -c "
from playwright.sync_api import sync_playwright
with sync_playwright() as p:
    b = p.chromium.launch(); c = b.new_context(storage_state='auth.json'); pg = c.new_page()
    pg.goto('https://dataghurhi.cse.buet.ac.bd/dashboard', wait_until='networkidle')
    print('signed in:', 'My Projects' in pg.inner_text('body')); b.close()"

# auth.json is a LIVE login — it is gitignored. Delete it when you are done.
