\chapter{Investigation and Evaluation}
\label{ch:evaluation}

\section{Experimental Investigation}
\label{sec:investigation}
\guidance{Present the investigation carried out to substantiate the
solution. State the research questions or hypotheses, the method and
experimental design, the datasets or measurements used and how they were
obtained, the baselines or points of comparison, and the metrics. Analyse
and interpret the data, and synthesise the findings into conclusions that
are supported by the evidence. Projects with a substantial research and
development component should treat this as the core of the chapter; for
development-focused projects, report the investigation that validated the
main design decisions. This is the experimental work on the built system;
the background research is in \Cref{ch:research}.}

\section{Performance Evaluation}
\label{sec:evaluation}
\guidance{Evaluate the system against every requirement and objective. Describe the
setup, tools, inputs, and acceptance criteria; present the results (tests,
measurements, security checks, user or partner acceptance, experiments as
appropriate); state which requirements are met and clearly explain each
deviation.}

\section{Deviations and Design Revisions}
\label{sec:revisions}
\guidance{For each deviation, give the technical analysis of its cause and the
design revision made to achieve compliance, or explain why it was not
revised.}
